% ---- 1. TRẮC NGHIỆM 4 PHƯƠNG ÁN: đúng MỘT \True
\begin{ex}%[2D1N1-1]
	Hàm số $y=x^3-3x$ nghịch biến trên khoảng nào?
	\choice
	{$(-\infty;-1)$}
	{\True $(-1;1)$}
	{$(1;+\infty)$}
	{$(-\infty;+\infty)$}
	\loigiai{
		$y'=3x^2-3<0\Leftrightarrow -1<x<1$.
	}
\end{ex}

% ---- 2. ĐÚNG/SAI: 4 ý, ý đúng có \True
\begin{ex}%[2D1H1-1]
	Cho hàm số $y=x^3-3x$. Các khẳng định sau đúng hay sai?
	\choiceTF
	{\True Hàm số có đạo hàm $y'=3x^2-3$}
	{Hàm số đồng biến trên $(-1;1)$}
	{\True Hàm số nghịch biến trên $(0;1)$}
	{Hàm số đồng biến trên $\mathbb{R}$}
	\loigiai{
		\begin{itemchoice}
			\itemch Đúng. $y'=3x^2-3$.
			\itemch Sai. Trên $(-1;1)$ ta có $y'<0$.
			\itemch Đúng. $(0;1)\subset(-1;1)$.
			\itemch Sai.
		\end{itemchoice}
	}
\end{ex}

% ---- 3. TRẢ LỜI NGẮN: môi trường bt + \shortans{đáp số}
\begin{bt}%[2D1V1-3]
	Có bao nhiêu giá trị nguyên của $m\in[-10;10]$ để hàm số $y=x^3-3mx+1$ đồng biến trên $\mathbb{R}$?
	\shortans{11}
	\loigiai{
		$y'=3x^2-3m\ge 0\ \forall x\Leftrightarrow m\le 0$, có $11$ giá trị nguyên.
	}
\end{bt}

% ---- 4. TỰ LUẬN: môi trường bt, không có \shortans
\begin{bt}%[2D1H1-1]
	Xét tính đơn điệu của hàm số $y=x^4-2x^2$.
	\loigiai{
		$y'=4x^3-4x$; hàm số nghịch biến trên $(-\infty;-1)$ và $(0;1)$, đồng biến trên $(-1;0)$ và $(1;+\infty)$.
	}
\end{bt}
