	Rút gọn biểu thức $H = (x + 2y)(x^2 - 2xy + 4y^2) - x^3$ rồi tính giá trị của $H$ tại $x = 2026$, $y = -2$.
	\loigiai{
		Ta có $H = (x^3 - 2x^2y + 4xy^2 + 2x^2y - 4xy^2 + 8y^3) - x^3 = 8y^3$. \\
Thay $y = -2$ vào $H$, ta được $H = 8 \cdot (-2)^3 = 8 \cdot (-8) = -64$.
	}
\end{bt}
