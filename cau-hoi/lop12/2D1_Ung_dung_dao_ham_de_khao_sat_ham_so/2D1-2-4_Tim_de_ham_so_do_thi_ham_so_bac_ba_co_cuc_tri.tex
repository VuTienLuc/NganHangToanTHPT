		\item $y = \dfrac{1}{3}x^3  - mx^2 + (m^2 - 4)x + 3$ đạt cực đại tại điểm $x = 3$.
	\end{enumerate}
	\loigiai{ 
		\begin{enumerate}
			\item Ta có $y' =  x^2 - 2mx + m^2 - 4$; $y'' = 2x - 2m$.\\
			Để hàm số đạt cực tiểu tại $x = 3$ khi \[\heva{& y'(3) = 0 \\ & y''(3)  > 0} \Leftrightarrow \heva{& m^2 - 6m + 5 = 0 \\ & 6 - 2m > 0} \Leftrightarrow \heva{& m = 1 \\ & m = 5 \Rightarrow m < 3} \Rightarrow m = 1.\]
			\item Có $y' = x^2 - 2mx + m^2 - m +  1,\ y'' = 2x - 2m$.\\
			Hàm số đạt cực đại tại $x = 1$ khi $\heva{& y'(1) = 0 \\  & y''(1) < 0} \Leftrightarrow \heva{& m^2 - 3m + 2 = 0 \\ & 2 - 2m < 0} \Rightarrow m = 2$.
			 \item Tập xác định $\mathscr{D} = \mathbb{R}$.\\
			Đạo hàm $y'  = -3x^2 + 2mx + m^2 - 12$, $y'' = -6x + 2m$.\\

			\item Ta có $y' = x^2 - 2mx +  (m^2 - 4)$; $y'' = 2x - 2m$.\\
			Hàm số đạt cực trị tại $x = 3$ suy ra $y'(3) = 0 \Rightarrow m^2 - 6m + 5 = 0 \Rightarrow  \hoac{& m = 1 \\ & m = 5}$.\\
			Với $m = 5$ ta có $y''(3) = 6 - 10 = -4  < 0$ suy ra hàm số đạt cực đại tại $x = 3$.\\
			Với $m = 1$ thì $y''(3) = 6 - 2 = 4 > 0$ suy ra hàm số  đạt cực tiểu tại $x = 3$.\\
			 Vậy $m = 5$ thì hàm số $y = \dfrac{1}{3}x^3 - mx^2 + (m^2 - 4)x + 3$ đạt cực đại tại điểm $x = 3$.
		\end{enumerate}
