\begin{ex}%[2H2H2-5]%[ma:ddbaea4c]
	Trong không gian $Oxyz$, cho các điểm $A(1;-3;  2)$, $B(2; 4;-2)$ và $C'(3; 2;-2)$.
	 \choiceTF
		 {\True Trung điểm của đoạn thẳng $O B$ là $C(1; 2;-1)$}
		{\True Biết rằng tứ giác $A C C' A'$  là hình bình hành. Cao độ của điểm $A'$ là $z=1$}
		{Biết rằng điểm $B'$ là  đỉnh còn lại của hình lăng trụ $A B C . A' B' C'$. Khi đó tung độ của điểm $B'$ là $y=-3$}
		{\True Thể tích của  khối lăng trụ $A B C . A' B' C'$ bằng $\dfrac{7}{2}$}
	\loigiai{ 
		\begin{itemchoice}
			 \itemch \textbf{Đúng}.
			Toạ độ  trung điểm $C\left(x_C; y_C; z_C\right)$ của đoạn thẳng $O B$ là
			$$\heva{&x_C=\dfrac{x_B+x_O}{2}  \\ &y_C=\dfrac{y_B+y_O}{2} \\& z_C=\dfrac{z_B+z_O}{2}} \Rightarrow\heva{&x_C=1 \\ &y_C=2 \\ &z_C=-1 .}$$
			Suy ra $C(1; 2;-1)$. 
			 \itemch \textbf{Đúng}.
			Vì $A$, $C$, $C'$ không thẳng hàng nên để tứ giác $A C C' A'$ là hình bình hành thì $$\overrightarrow{A A'}=\overrightarrow{C  C'}.$$
			Suy ra $A'(3;-3; 1)$, do đó cao  độ của điểm $A'$ là $z=1$.
			\itemch  \textbf{Sai}.
			Vì điểm $B'$ là đỉnh còn lại của hình lăng trụ $A B C.A'  B' C'$ nên $\overrightarrow{B B'}=\overrightarrow{C C'}$.\\
			Suy ra $B'(4; 4;-3)$. Do  đó tung độ của điểm $B'$ là $y=4$.
			\itemch \textbf{Đúng}. 
			Ta có $V_{A B C. A' B' C'}=S_{A B C} \cdot  \mathrm{d}\left(A',(A B C)\right)$.
			 $$
			\left[\overrightarrow{A B}, \overrightarrow{A C}\right]=(-1; 3; 5) \text {nên  } S_{A B C}=\dfrac{1}{2}\left|\left[\overrightarrow{A B}, \overrightarrow{A C}\right]\right|=\dfrac{\sqrt{35}}{2}.
			 $$
			Phương trình mặt phẳng đi qua ba  điểm $A$, $B$, $C$ là $x-3 y-5 z=0$. Khi đó
			$$ 
			\mathrm{d}\left(A',(A B C)\right)=\dfrac{|3-3 \cdot(-3)-5 \cdot 1|}{\sqrt{35}}=\dfrac{7  \sqrt{35}}{35}.$$
			Vậy $V_{A  B C . A' B' C'}=\dfrac{7}{2}$.
		\end{itemchoice}
	} 
\end{ex}

		Suy ra $A$, $B$ đối xứng với nhau qua điểm $I(0;2)$ là tâm đối xứng của  $(C)$.\\
		$AB\perp  d\colon x+y-5=0\Rightarrow AB\colon x-y+m=0$.\\
		$AB$ qua  $I$ nên ta có $m=2\Rightarrow AB\colon x-y+2=0$.\\
		Khi đó hoành độ  $A$, $B$ thỏa mãn phương trình
		$x^{3}-3x+2=x+2\Leftrightarrow \hoac{&x=0\,(\text{loại})  \\
		 &x=\pm 2.}$\\
		$\Rightarrow  A(2;4)$, $B(-2;0)$.\\
		Do đó, $b=4$  và $d=0$.\\
		Vậy $b+d=4+0=4$. 
	 }
\end{bt}
