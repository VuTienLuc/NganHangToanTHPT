\textbf{1. Kiến thức cần nhớ}
\begin{itemize}
\item \textbf{Độ dài đường tròn (chu vi)}: $\boxed{C = 2\pi R = \pi d}$ ($R$ là bán kính, $d$ là đường kính).
\item \textbf{Độ dài cung tròn} $n^\circ$:
$$\boxed{l = \dfrac{\pi R n}{180}}$$
\item \textbf{Diện tích hình tròn}: $\boxed{S = \pi R^2 = \dfrac{\pi d^2}{4}}$.
\item \textbf{Diện tích hình quạt tròn} bán kính $R$, cung $n^\circ$:
$$\boxed{S_q = \dfrac{\pi R^2 n}{360} = \dfrac{l \cdot R}{2}}$$
(với $l$ là độ dài cung của hình quạt tròn đó).
\item \textbf{Diện tích hình vành khuyên} giới hạn bởi hai đường tròn đồng tâm bán kính $R$ và $r$ ($R > r$):
$$\boxed{S_{vk} = \pi (R^2 - r^2)}$$
\end{itemize}

\textbf{2. Phương pháp giải các dạng}

%% DANG 9H5?15-1
\textbf{Dạng 1. Độ dài của đường tròn, cung tròn}
\begin{enumerate}
\item Xác định bán kính $R$ và số đo góc ở tâm $n^\circ$ chắn cung đó.
\item Thay vào công thức $l = \dfrac{\pi R n}{180}$.
\item Khi biết độ dài cung $l$, muốn tính số đo độ $n^\circ$: $\boxed{n = \dfrac{180 \cdot l}{\pi R}}$.
\end{enumerate}

%% DANG 9H5?15-2
\textbf{Dạng 2. Diện tích hình tròn, hình quạt tròn}
\begin{enumerate}
\item Áp dụng công thức $S = \pi R^2$ và $S_q = \dfrac{\pi R^2 n}{360}$.
\item Tính diện tích hình viên phân: Lấy diện tích hình quạt tròn trừ đi diện tích tam giác tương ứng:
$$S_{\text{viên phân}} = S_q - S_{\triangle OAB}.$$
\end{enumerate}

%% DANG 9H5?15-3
\textbf{Dạng 3. Diện tích hình vành khuyên, bài toán thực tế}
\begin{enumerate}
\item Xác định bán kính vòng ngoài $R$ và bán kính vòng trong $r$ (lưu ý bề rộng viền $d = R - r$).
\item Áp dụng công thức $S_{vk} = \pi (R^2 - r^2)$.
\item Bài toán thực tế: Lối đi xung quanh hồ bơi hình tròn, đường chạy sân vận động, mặt cắt của ống nước, diện tích phủ sơn\dots
\end{enumerate}
